2. Moteur principal du jeu

Fichier: src/game/engine.js

C’est l’extrait le plus fort du projet. Il illustre la séparation entre état métier, physique du joueur, runtime technique et services comme le renderer ou la vue. 

// Engine is the orchestration layer of the game.
// It owns the main loop, coordinates state updates and tells the view/renderer what to show.
export function createGame(elements) {
  const { browser, canvas, view } = elements;

  const ctx = canvas.getContext("2d");
  const world = createWorld();
  const renderer = createRenderer(ctx, world);

  canvas.width = world.width;
  canvas.height = world.height;

  let difficulty = getDifficultyById(loadPreferredDifficulty());
  // Long-lived simulation state.
  const state = createGameState(difficulty, loadBestScore(difficulty.id));
  // Player physics state.
  const player = createPlayer(world, difficulty);
  // Ephemeral runtime bookkeeping for the loop and asset loading.
  const runtime = createRuntimeState();

  let obstacles = [];
  let bonusHearts = [];
  let bugHitEffects = [];
  let clouds = [];

  function initGame() {
    view.clearHeartBreakTimers();
    clearHeadImpactTimer();
    obstacles = [];
    bonusHearts = [];
    bugHitEffects = [];
    clouds = createClouds();
    resetRuntimeState(runtime);
    resetGameState(state, difficulty, loadBestScore(difficulty.id));
    resetPlayer(player, difficulty);

    // Resetting the simulation also resets the UI so the next run starts from a clean baseline.
    view.resetLivesDisplay(state.lives);
    updateScoreDisplay();
    view.showOverlay("Run & Debug", getJumpInstructions(), "Commencer");
    syncTouchMode();
    renderer.draw({ state, player, runtime, clouds, obstacles, bonusHearts, bugHitEffects });
  }

3. Boucle de gameplay pure et testable

Fichier: src/game/gameplay-loop.js

Ici, la logique de frame est isolée dans des fonctions pures. C’est une bonne preuve de qualité d’architecture, parce qu’on peut tester la logique sans dépendre du DOM ni du canvas. 

import { computeNextObstacleDelay } from "./logic.js";

// Pure frame-update helpers extracted from the engine.
// The engine orchestrates them, while tests can validate the math in isolation.
export function computeFrameDelta(timestamp, lastTime, maxDelta = 0.033) {
  if (!lastTime) {
    return {
      deltaTime: 0,
      lastTime: timestamp,
    };
  }

  return {
    deltaTime: Math.min((timestamp - lastTime) / 1000, maxDelta),
    lastTime: timestamp,
  };
}

export function createActiveGameUpdate(state, difficulty, worldWidth, deltaTime) {
  // Once the death fall starts we freeze score/speed progression,
  // but we keep animation timers advancing until the ending resolves.
  const isEndingAfterAnimation = state.pendingGameOver;
  const runTime = isEndingAfterAnimation ? state.runTime : state.runTime + deltaTime;
  const score = isEndingAfterAnimation ? state.score : state.score + deltaTime * difficulty.scoreRate;
  const speed = isEndingAfterAnimation
    ? state.speed
    : state.speed + deltaTime * (difficulty.speedRamp + runTime * difficulty.speedRampGrowth);

  return {
    runTime,
    score,
    speed,
    obstacleTimer: isEndingAfterAnimation ? state.obstacleTimer : state.obstacleTimer + deltaTime,
    hitCooldown: isEndingAfterAnimation
      ? state.hitCooldown
      : Math.max(0, state.hitCooldown - deltaTime),
    groundOffset: (state.groundOffset + speed * deltaTime) % worldWidth,
  };
}

4. Couche UI séparée de la logique métier

Fichier: src/game/ui.js

Cet extrait montre que l’interface ne décide pas des règles du jeu. Elle se contente de refléter l’état et d’animer le DOM, ce qui rend le code plus lisible et plus maintenable. 

// DOM-only presentation layer.
// It updates overlays, score panels and heart animations, but never runs game rules.
export function createUI({
  overlay,
  overlayTitle,
  overlayText,
  actionButton,
  recordValue,
  scoreValue,
  livesDisplay,
  difficultyOptions,
  touchJumpButton,
}, browser) {
  const heartBreakTimers = new Map();

  function setupLivesDisplay(maxLives, lives) {
    livesDisplay.innerHTML = "";

    // Hearts are generated once, then their CSS classes reflect the current state.
    for (let index = 0; index < maxLives; index += 1) {
      const heart = browser.documentRef.createElement("span");
      heart.className = "heart";
      heart.dataset.index = String(index);
      heart.setAttribute("aria-hidden", "true");
      livesDisplay.appendChild(heart);
    }

    updateLivesAriaLabel(lives);
  }

  function breakHeart(index, lives) {
    const heart = livesDisplay.querySelector(`.heart[data-index="${index}"]`);
    if (!heart) {
      updateLivesAriaLabel(lives);
      return;
    }

    heart.classList.add("heart--breaking");
    // The engine updates lives immediately; this timeout only drives the visual transition.
    const timer = browser.setTimeout(() => {
      heart.classList.remove("heart--breaking");
      heart.classList.add("heart--gone");
      heartBreakTimers.delete(index);
      updateLivesAriaLabel(lives);
    }, 450);

    heartBreakTimers.set(index, timer);
  }

import test from "node:test";
import assert from "node:assert/strict";

import { createGameState } from "../src/game/game-state.js";

  computeFrameDelta,
  createActiveGameUpdate,
} from "../src/game/gameplay-loop.js";

test("player controller allows one jump and one double jump", () => {
  const difficulty = {
    startSpeed: 320,
    maxSpeedFactor: 1.35,
    obstacleDelayMin: 0.9,
    obstacleDelayMax: 1.55,
    gravity: 2100,
    jumpStrength: 760,
    doubleJumpMultiplier: 0.94,
  };
  const world = { groundY: 430 };
  const state = createGameState(difficulty, 0);
  const player = createPlayer(world, difficulty);

  state.gameStarted = true;

  assert.equal(jumpPlayer(state, player), true);
  assert.equal(player.isJumping, true);
  assert.equal(player.jumpsUsed, 1);

  assert.equal(jumpPlayer(state, player), true);
  assert.equal(player.isDoubleJumping, true);
  assert.equal(player.jumpsUsed, 2);
});
